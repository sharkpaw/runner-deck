{{/* runnerRows: the  content of the list. The first render is expanded directly by the page; on auto-refresh,
     /api/runner-rows renders the same fragment separately. Both paths share one row-rendering logic,
     so three-state checks like GitHubYes/GitHubNo don't need to be rewritten in JS. */}}
{{define "runnerRows"}}
        {{range .Runners}}
        {{/* Filtering and counting read these attributes instead of parsing the visible text in the cells. That text is
             translated per language, so filtering would break after switching languages. The attributes live in the fragment, so auto-refresh
             returned rows carry them too. */}}
        
          {{.Name}}
          {{.TargetType}}: {{.Target}}
          
            {{with index $.T (printf "status.%s" .Status)}}{{.}}{{else}}{{.Status}}{{end}}
            {{if .Running}}{{index $.T "badge.running"}}{{end}}
            {{/* Use GitHubBusyYes instead of {{if .GitHubBusy}}: it is also a *bool,
                 and a pointer to false is also truthy in templates, so "idle" would render as "busy" */}}
            {{if .GitHubBusyYes}}{{index $.T "badge.busy"}}{{index $.T "badge.busy_title"}}{{end}}
            {{/* Likewise use GitHubPublicYes: {{if .GitHubPublic}} would attach this badge to every private repo */}}
            {{if .GitHubPublicYes}}{{index $.T "badge.public_repo"}}{{index $.T "badge.public_repo_title"}}{{end}}
            {{if .ContainerDrift}}{{index $.T "badge.drift"}}{{.ContainerDrift}}{{end}}
            {{if .Probe}}
{{index $.T "probe.failed"}}{{if .Probe.Type}} ({{.Probe.Type}}){{end}}{{.Probe.Error}}{{end}}
          
          {{if $.Config.Runners.ContainerMode}}{{.JobDockerBackend}}{{end}}
          
            {{if .RegistrationMessage}}
              {{if and (eq .Status "installed") .GitHubURL}}
                {{/* The tooltip still shows the raw registration result. That is the real diagnostic for this cell, and turning it into a link should not crowd it out.
                     "Clicking goes to GitHub" is conveyed by the dotted underline and the GitHub ✓ on the next line */}}
                {{index $.T "reg.registered"}}{{.RegistrationMessage}}
              {{else}}
                {{if eq .Status "installed"}}{{index $.T "reg.registered"}}{{else}}{{index $.T "reg.failed"}}{{end}}{{.RegistrationMessage}}
              {{end}}
            {{else}}
              —
            {{end}}
            {{/* Three-state values must use GitHubYes/GitHubNo: {{if .RegisteredOnGitHub}} only checks whether a pointer is nil,
                 so a pointer to false is also truthy, and "not on GitHub" would render as "GitHub ✓" */}}
            {{if .GitHubYes}}
              
{{if .GitHubURL}}{{index $.T "github.yes"}}{{else}}{{index $.T "github.yes"}}{{end}}
            {{else if .GitHubNo}}
              
{{index $.T "github.no"}}
